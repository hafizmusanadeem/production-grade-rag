===== COMPARE.PY =====
"""evals/compare.py -- the decision function.

Reads two snapshots (baseline, candidate) written by run_suite.py, looks up each
metric's rule in the registry, classifies every metric, returns PASS/REVIEW/FAIL.
Exit code makes it a CI gate: PASS=0, FAIL=1, REVIEW=2.
"""
import argparse
import json
import math
import sys
from collections import Counter

from evals.metric_registry import rule_for

BASELINE_PATH = "baselines/baseline.json"
CANDIDATE_PATH = "baselines/candidate.json"

EXIT_CODE = {"PASS": 0, "FAIL": 1, "REVIEW": 2}


def load(path):
    with open(path) as f:
        return json.load(f)


def _is_number(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool) \
        and not (isinstance(x, float) and math.isnan(x))


def classify(mid, bv, cv, rule):
    row = {"id": mid, "baseline": bv, "candidate": cv,
           "kind": rule["kind"], "direction": rule["direction"],
           "delta": None, "status": None}

    if bv is None:
        row["status"] = "new"
        return row
    if cv is None:
        row["status"] = "dropped"
        return row

    if rule["kind"] == "info":
        if _is_number(bv) and _is_number(cv):
            row["delta"] = cv - bv
        row["status"] = "info"
        return row

    # boolean SLO flags
    if rule.get("bool") or isinstance(bv, bool) or isinstance(cv, bool):
        if bv == cv:
            row["status"] = "flat"
        elif (not bv) and cv:
            row["status"] = "improved"
        else:
            row["status"] = ("blocked" if rule["kind"] == "gate" else "regressed")
        return row

    if not (_is_number(bv) and _is_number(cv)):
        row["status"] = "info"
        return row

    delta = cv - bv
    row["delta"] = delta
    higher_better = rule["direction"] == "higher"
    improved = (delta > 0) if higher_better else (delta < 0)
    if improved:
        row["status"] = "improved"
        return row

    worse = abs(delta)
    tolerance = max(rule["tol"], rule["rel_tol"] * abs(bv))
    if worse <= tolerance:
        row["status"] = "flat"
    else:
        row["status"] = ("blocked" if rule["kind"] == "gate" else "regressed")
    return row


def compare(baseline, candidate):
    b = baseline.get("metrics", {})
    c = candidate.get("metrics", {})

    rows = [classify(mid, b.get(mid), c.get(mid), rule_for(mid))
            for mid in sorted(set(b) | set(c))]

    if any(r["status"] == "blocked" for r in rows):
        verdict = "FAIL"
    elif any(r["status"] == "regressed" for r in rows):
        verdict = "REVIEW"
    else:
        verdict = "PASS"
    return verdict, rows


def _fmt(v):
    if isinstance(v, bool):
        return str(v)
    if isinstance(v, float):
        return "nan" if math.isnan(v) else f"{v:.4g}"
    return str(v)


def print_report(verdict, rows, show_all=False):
    order = {"blocked": 0, "regressed": 1, "dropped": 2, "new": 3,
             "improved": 4, "flat": 5, "info": 6}
    shown = [r for r in rows if show_all or r["kind"] != "info"]
    shown.sort(key=lambda r: (order.get(r["status"], 9), r["id"]))

    print("=" * 96)
    print(f"{'metric':<46} {'baseline':>10} {'candidate':>10} {'delta':>10}  status")
    print("-" * 96)
    for r in shown:
        d = r["delta"]
        dstr = f"{d:+.4g}" if isinstance(d, (int, float)) and not isinstance(d, bool) else ""
        marker = "  <<<" if r["status"] in ("blocked", "regressed") else ""
        print(f"{r['id']:<46} {_fmt(r['baseline']):>10} {_fmt(r['candidate']):>10} "
              f"{dstr:>10}  {r['status']}{marker}")

    counts = Counter(r["status"] for r in rows)
    line = "  ".join(f"{k}={counts[k]}" for k in
                     ["blocked", "regressed", "improved", "flat", "new", "dropped", "info"]
                     if counts[k])
    print("-" * 96)
    print(line)
    info_n = counts["info"]
    if not show_all and info_n:
        print(f"({info_n} info metrics hidden; pass --all to show them)")
    print("=" * 96)

    banner = {
        "PASS":   "PASS   -- no gate blocked, no guardrail regressed. Safe to promote.",
        "REVIEW": "REVIEW -- a guardrail regressed beyond tolerance. A human decides.",
        "FAIL":   "FAIL   -- a gate regressed. Blocked.",
    }[verdict]
    print(f"VERDICT: {banner}")
    print("=" * 96)


def main():
    ap = argparse.ArgumentParser(description="Compare two eval snapshots.")
    ap.add_argument("--baseline", default=BASELINE_PATH)
    ap.add_argument("--candidate", default=CANDIDATE_PATH)
    ap.add_argument("--all", action="store_true", help="also show info metrics")
    args = ap.parse_args()

    baseline = load(args.baseline)
    candidate = load(args.candidate)

    bl = baseline.get("metadata", {})
    cl = candidate.get("metadata", {})
    label_b = bl.get("label") or args.baseline
    label_c = cl.get("label") or args.candidate
    git_b = bl.get("git_sha", "?")
    prompt_b = bl.get("prompt_hash", "?")
    print(f"baseline : {label_b}   ({git_b}, prompt {prompt_b})")
    print(f"candidate: {label_c}   ({cl.get('git_sha', '?')}, prompt {cl.get('prompt_hash', '?')})")

    verdict, rows = compare(baseline, candidate)
    print_report(verdict, rows, show_all=args.all)
    sys.exit(EXIT_CODE[verdict])


if __name__ == "__main__":
    main()


===== RUN_SUITE.PY =====
"""evals/run_suite.py -- the orchestrator.

Runs the FULL eval suite against ONE pipeline (built once and injected everywhere),
flattens every metric into dotted ids, stamps metadata, writes a snapshot JSON.
The snapshot is the atomic unit regression testing compares.

    python -m evals.run_suite                          -> baselines/candidate.json
    python -m evals.run_suite --baseline               -> baselines/baseline.json
    python -m evals.run_suite --label "k=10 + reranker"  (describe the change)
    python -m evals.run_suite --quiet                  -> suppress ops/safety chatter
"""
import argparse
import hashlib
import json
import os
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

from dotenv import load_dotenv

from app.generator import build_prompt
from evals.metric_registry import rule_for
from evals.eval_files import eval_retriever, eval_generator, eval_rag_pipeline, eval_application, eval_ops

load_dotenv(override=True)

BASELINE_PATH = "baselines/baseline.json"
CANDIDATE_PATH = "baselines/candidate.json"

SUITE = ["retriever", "generator", "pipeline", "application", "safety", "ops"]


def _git_sha():
    try:
        return subprocess.check_output(
            ["git", "rev-parse", "--short", "HEAD"], text=True,
            stderr=subprocess.DEVNULL,
        ).strip()
    except Exception:
        return "unknown"


def _prompt_hash():
    return hashlib.sha256(str(build_prompt("test query", [], intent="technical")).encode()).hexdigest()[:12]


def build_metadata(label):
    return {
        "created_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "git_sha": _git_sha(),
        "prompt_hash": _prompt_hash(),
        "label": label,
    }


def _slug(name):
    return name.strip().lower().replace(" ", "_")


def flatten_nested(namespace, summary):
    out = {}
    for metric, stats in summary.items():
        slug = _slug(metric)
        for stat, val in stats.items():
            out[f"{namespace}.{slug}.{stat}"] = val
    return out


def prefix_flat(namespace, flat):
    return {f"{namespace}.{k}": v for k, v in flat.items()}


def run_suite(label="", quiet=False, full=False):
    """Run all suites against one agent target; return a snapshot dict."""
    verbose = not quiet
    t0 = time.perf_counter()

    print("Building AgentTarget (once)...")
    agent_target = None  # we use eval_ops/eval_safety which create their own; or pass

    metrics = {}

    # retriever (uses RetrievalPipeline directly)
    print("\n[1] retriever eval...")
    metrics.update(flatten_nested("retriever", eval_retriever.run_local()))

    # generator (isolated, golden context)
    print("\n[2] generator eval...")
    metrics.update(flatten_nested("generator", eval_generator.run()))

    # RAG triad via AgentTarget
    print("\n[3] rag_pipeline (triad) eval...")
    metrics.update(flatten_nested("pipeline", eval_rag_pipeline.run(agent_target)))

    # application correctness/completeness via AgentTarget
    print("\n[4] application eval...")
    metrics.update(flatten_nested("application", eval_application.run(agent_target)))

    # safety
    print("\n[5] safety evals...")
    metrics.update(prefix_flat("safety", eval_ops_safety()))

    # ops
    print("\n[6] operational evals...")
    ops_snapshot = eval_ops.run_ops(verbose=verbose)
    metrics.update({f"ops.{k}": v for k, v in ops_snapshot.items()})

    elapsed = time.perf_counter() - t0

    if not full:
        metrics = {mid: v for mid, v in metrics.items() if rule_for(mid)["kind"] != "info"}

    # Stamp operational sub-keys under their proper namespace
    new_ops = {}
    for k, v in ops_snapshot.items():
        new_key = f"ops.{k}"
        new_ops[new_key] = v
    final = dict(metrics)
    final.update(new_ops)

    return {
        "metadata": {
            **build_metadata(label),
            "suite_seconds": round(elapsed, 1),
            "full": full,
            "n_metrics": len(metrics),
        },
        "metrics": final,
    }


def eval_ops_safety():
    """Wrapper to call the safety suite under the 'safety' namespace."""
    from evals.eval_files.eval_safety import run_safety
    return run_safety(verbose=True)


def write_snapshot(snapshot, path):
    os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
    with open(path, "w") as f:
        json.dump(snapshot, f, indent=2, sort_keys=True)
    return path


def print_snapshot(snapshot):
    meta = snapshot["metadata"]
    metrics = snapshot["metrics"]
    print("\n" + "=" * 74)
    print("SNAPSHOT")
    print("=" * 74)
    print(f"label       : {meta.get('label') or '(none)'}")
    print(f"created_at  : {meta['created_at']}")
    print(f"git_sha     : {meta['git_sha']}    prompt_hash: {meta['prompt_hash']}")
    print(f"suite_time  : {meta['suite_seconds']}s     metrics: {len(metrics)}")
    print("-" * 74)
    for key in sorted(metrics):
        val = metrics[key]
        shown = f"{val:.4f}" if isinstance(val, float) else str(val)
        print(f"  {key:<44} {shown}")
    print("=" * 74)


def main():
    ap = argparse.ArgumentParser(description="Run the full eval suite.")
    ap.add_argument("--baseline", action="store_true", help=f"write to {BASELINE_PATH}")
    ap.add_argument("--out", default=None)
    ap.add_argument("--label", default="")
    ap.add_argument("--quiet", action="store_true")
    ap.add_argument("--full", action="store_true", help="also write info metrics")
    args = ap.parse_args()

    out = args.out or (BASELINE_PATH if args.baseline else CANDIDATE_PATH)

    snapshot = run_suite(label=args.label, quiet=args.quiet, full=args.full)
    print_snapshot(snapshot)
    path = write_snapshot(snapshot, out)
    print(f"\nwrote snapshot -> {path}")


if __name__ == "__main__":
    main()


===== METRIC_REGISTRY.PY =====
"""evals/metric_registry.py -- the metric registry.

For every metric the suite produces, three questions decide how a change is
judged: direction, kind (gate/guardrail/info), tolerance. The keys in this file
are NOT public API --- rule_for(metric_id) is the single entry point evals and
compare.py use. Do not change the rules here unless you also update P8.

Reading this file: judge metrics are gated on their AVERAGE SCORE (0-1 scale)
not pass_rate -- see docstring at top. Operational metrics are direct
measurements (percentiles, cost, rates) with unchanged rules.
"""
import json
from pathlib import Path

GOLDEN_DIR = Path(__file__).resolve().parent.parent / "goldens"

# Presets ----------------------------------------------------------
# tol     = absolute tolerance, in the metric's own units
# rel_tol = relative tolerance, as a fraction of |baseline|
# A regression counts only if the worsening exceeds max(tol, rel_tol * |baseline|).

# Gate on average score (0-1): any drop BLOCKS the release. Safety lives here.
GATE_HIGHER_AVG = {"direction": "higher", "kind": "gate",      "tol": 0.02, "rel_tol": 0.0}
GATE_LOWER_AVG  = {"direction": "lower",  "kind": "gate",      "tol": 0.02, "rel_tol": 0.0}
QUALITY_GUARD   = {"direction": "higher", "kind": "guardrail", "tol": 0.05, "rel_tol": 0.0}   # quality

LATENCY_GUARD   = {"direction": "lower",  "kind": "guardrail", "tol": 0.0,  "rel_tol": 0.25}   # 25%
COST_GUARD      = {"direction": "lower",  "kind": "guardrail", "tol": 1e-6, "rel_tol": 0.15}  # 15% (abs floor)
SUCCESS_GUARD   = {"direction": "higher", "kind": "guardrail", "tol": 1.0,  "rel_tol": 0.0}
ERROR_GUARD     = {"direction": "lower",  "kind": "guardrail", "tol": 1.0,  "rel_tol": 0.0}
SLO_BOOL_GUARD  = {"direction": "higher", "kind": "guardrail", "tol": 0.0,  "rel_tol": 0.0, "bool": True}
INFO            = {"direction": "higher", "kind": "info",      "tol": 0.0,  "rel_tol": 0.0}

# The operational number that drives the verdict. Everything else under ops.*
# (p50/p99/mean, token counts, monthly total, INR, request count) is info.
LATENCY_GUARDED = {"ops.latency.e2e_p95_ms"}


def rule_for(metric_id: str) -> dict:
    mid = metric_id

    # ---- SAFETY GATES (on average score) ----
    if mid == "safety.toxicity.avg_toxicity":
        return GATE_LOWER_AVG
    if mid.startswith("safety.") and mid.endswith("avg_score"):
        return GATE_HIGHER_AVG

    # ---- QUALITY GUARDRAILS (on avg score) ----
    if mid.endswith("avg_score"):
        return QUALITY_GUARD

    # ---- OPERATIONAL ----
    if mid in LATENCY_GUARDED:
        return LATENCY_GUARD
    if mid == "ops.cost.cost_per_query_usd":
        return COST_GUARD
    if mid == "ops.reliability.success_rate":
        return SUCCESS_GUARD
    if mid == "ops.reliability.error_rate":
        return ERROR_GUARD
    if mid.endswith("_pass"):
        return SLO_BOOL_GUARD

    return INFO


===== RETRIEVAL_METRICS.PY =====
"""evals/eval_files/retrieval_metrics.py -- deterministic retrieval metrics.

These are cheap regression signals that don't need an LLM judge (no cost, ~1s runtime).
They key off each golden's source file_name: did the retrieved chunk come from the
golden's expected source? source_hit_at_k is 1 if any retrieved doc matches.
source_mrr ranks the chunks by original score and reports mean reciprocal rank
of hits from the golden's source. source_precision_at_k checks whether the top-k
retrieved docs included the golden's source at all.
"""
from typing import List

GOLDEN_PATH = "evals/goldens/retriever.json"
RETRIEVER_PATH = "evals/eval_files/eval_retriever.py"


def _load_goldens(path):
    import json
    with open(path, encoding="utf-8") as f:
        return json.load(f)


def run(retriever_invoke, *, limit: int = None, ids: List[str] = None) -> dict:
    """deterministic metrics for a single query -> list of RetrievedChunk pairs."""
    if limit is None:
        limit = 50
    ids = ids or []
    goldens = _load_goldens(GOLDEN_PATH)
    results = {}
    for g in goldens[:limit]:
        docs = retriever_invoke(g["query"])
        doc_list = list(docs)
        # hit_at_k: any retrieved doc whose file_name == golden.source
        hits = [d for d in doc_list if d.metadata and d.metadata.get("file_name") == g["source"]]
        results[g["id"]] = {
            "hit_at_k": len(hits),
            "score": sum(d.score for d in doc_list) / len(doc_list) if doc_list else 0.0,
            "k": len(doc_list),
        }
    # Aggregate
    n = len(results)
    hit_vals = [v["hit_at_k"] for v in results.values()]
    score_vals = [v["score"] for v in results.values()]
    prec_vals = [(v["hit_at_k"] + 0) / max(v["k"], 1) for v in results.values()]

    out = {
        "contextual_recall_avg": sum(hit_vals) / n if n else 0.0,
        "contextual_precision_avg": sum(prec_vals) / n if n else 0.0,
        "source_hit_at_k": sum(hit_vals) / n if n else 0.0,
        "source_mrr": (sum(1 / max(i + 1, 1) for i in enumerate(sorted([h for h in hit_vals])) ) / max(n, 1)),
        "pass_rate": 100 * sum(1 for h in hit_vals if h >= 1) / n,
        "n": n,
    }
    return out


def compare():
    """base vs production: run both retrievers on a sample and report hit_at_k."""
    import random
    judge = None  # this is just a helper
    goldens = _load_goldens(GOLDEN_PATH)
    sample = random.sample(goldens, min(20, len(goldens)))

    from app.services.retrieval.retrieval_pipeline import RetrievalPipeline, QueryRequest
    from app.config import RETRIEVAL_LIMIT_K, RERANK_TOP_N, settings
    base_pipe = RetrievalPipeline()
    prod_pipe = RetrievalPipeline()

    results = {"base": [], "production": []}
    for g in sample:
        req = QueryRequest(query=g["query"], limit_k=RETRIEVAL_LIMIT_K, top_n=RERANK_TOP_N,
                           use_reranker=False)
        resp = base_pipe.process_query(req)
        docs = resp.results
        hits = [d for d in docs if d.metadata and d.metadata.get("file_name") == g["source"]]
        results["base"].append(len(hits))

        req2 = QueryRequest(query=g["query"], limit_k=RETRIEVAL_LIMIT_K, top_n=RERANK_TOP_N,
                             use_reranker=True, min_rerank_score=settings.RERANK_MIN_SCORE)
        resp2 = prod_pipe.process_query(req2)
        docs2 = resp2.results
        hits2 = [d for d in docs2 if d.metadata and d.metadata.get("file_name") == g["source"]]
        results["production"].append(len(hits2))

    print("Deterministic retrieval comparison (sample %d):" % len(sample))
    print(f"{'id':<8} {'base':>6} {'prod':>6} {'delta':>7}")
    deltas = []
    for i, g in enumerate(sample):
        bv, pv = results["base"][i], results["production"][i]
        delta = pv - bv
        deltas.append(delta)
        print(f"{g['id']:<8} {bv:>6} {pv:>6} {delta:>+7}")
    print(f"\nbase avg hit_at_k: {sum(results['base']) / len(results['base']):.3f}")
    print(f"production avg hit_at_k: {sum(results['production']) / len(results['production']):.3f}")
    print(f"delta avg: {sum(deltas) / len(deltas):+.3f}")
